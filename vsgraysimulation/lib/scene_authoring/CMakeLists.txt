cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(scene_deployment LANGUAGES CXX)
endif()
if(NOT TARGET nlohmann_json::nlohmann_json)
    find_package(nlohmann_json CONFIG REQUIRED)
endif()
add_library(scene_deployment STATIC Deployment.cpp)
target_compile_features(scene_deployment PUBLIC cxx_std_17)
target_include_directories(scene_deployment PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(scene_deployment PUBLIC nlohmann_json::nlohmann_json)
